% Inline code — breakable at path/command/JSON separators, so long inline paths
% and commands wrap instead of overflowing. Spaces and nested macros (e.g.
% \textit placeholders) are preserved; \allowbreak is only an opportunity, so
% short code never breaks.
\ExplSyntaxOn
\tl_new:N \l__code_tl
\cs_new_protected:Npn \__code_break:n #1 {
  \tl_set:Nn \l__code_tl {#1}
  \regex_replace_all:nnN { [/.,:;"=\[\]-] | \c{_} } { \0 \c{allowbreak} } \l__code_tl
  \tl_use:N \l__code_tl
}
\NewDocumentCommand \code { m } { \texttt { \color{inkdark} \__code_break:n {#1} } }
\ExplSyntaxOff
